アプリ独自のスタイルには QuickGUI オーバーレイ、ファイル選択やアラートにはネイティブダイアログを使います。

## Popover

Popover はトリガー、Positioner、Popup を組み合わせます。指定した方向は優先設定で、コアが衝突を解決します。ルート内の `ui.UsePopoverPlacement` で確定位置を取得できます。

```go
func PopoverExample() *native.Node {
	popover1 := ui.NewPopover(ui.PopoverRootProps{})
	return popover1.Root().Children(func() *native.Node {
		return ui.Fragment([]*native.Node{popover1.Trigger(ui.PopoverTriggerProps{}).Children("Account").NativeNode(), popover1.Positioner(ui.PopoverPositionerProps{}).Children(func() *native.Node {
			return popover1.Popup(ui.PopoverPopupProps{PartProps: ui.PartProps{Style: ui.Style().Width(240).Padding(16)}}).Children(func() *native.Node {
				return ui.Fragment([]*native.Node{popover1.Title(ui.PartProps{}).Children("Account").NativeNode(), popover1.Close(ui.PartProps{}).Children("Done").NativeNode()})
			}).NativeNode()
		}).NativeNode()})
	}).NativeNode()
}
```

## Dialog

モーダル内容は `Dialog.Portal` の下で作成します。`Dialog.Title` と `Dialog.Description` がアクセシブルな名前と説明を提供します。`AlertDialog` は確認の意味を加えます。制御されたルートは次の開閉状態とその理由を受け取ります。

```go
func DialogExample() *native.Node {
	dialog1 := ui.NewDialog(ui.DialogRootProps{})
	return dialog1.Root().Children(func() *native.Node {
		return ui.Fragment([]*native.Node{dialog1.Trigger(ui.PartProps{}).Children("Open dialog").NativeNode(), dialog1.Portal(ui.PartProps{}).Children(func() *native.Node {
			return ui.Fragment([]*native.Node{dialog1.Backdrop(ui.PartProps{}).NativeNode(), dialog1.Viewport(ui.PartProps{}).Children(func() *native.Node {
				return dialog1.Popup(ui.DialogPopupProps{}).Children(func() *native.Node {
					return ui.Fragment([]*native.Node{dialog1.Title(ui.PartProps{}).Children("Confirm action").NativeNode(), dialog1.Description(ui.PartProps{}).Children("Review the details before continuing.").NativeNode(), dialog1.Close(ui.PartProps{}).Children("Cancel").NativeNode()})
				}).NativeNode()
			}).NativeNode()})
		}).NativeNode()})
	}).NativeNode()
}
```

## システムポップオーバー

`ui.SystemPopover` は OS のポップオーバー内に内容をホストします。論理幅と高さを指定し、通常の QuickGUI の子を宣言します。

```go
func SystemPopoverExample() *native.Node {
	systemPopover1 := ui.NewSystemPopover(ui.PopoverRootProps{})
	return systemPopover1.Root().Children(func() *native.Node {
		return ui.Fragment([]*native.Node{systemPopover1.Trigger(ui.PopoverTriggerProps{}).Children("Account").NativeNode(), systemPopover1.Content(ui.PopoverContentProps{Width: 240, Height: 120, Placement: "bottom-start"}).Children("Account settings").NativeNode()})
	}).NativeNode()
}
```

## ネイティブダイアログ

ネイティブダイアログは非同期コールバックを使います。レンダリング時に親ウィンドウを取得してリクエストへ明示的に渡します。キャンセルされた場合は `Canceled: true` が返ります。

```go
func OpenFolderButton() *ui.Element {
	window := native.CurrentWindow()
	return ui.Button().Child("Open folder").OnClick(func() {
		native.ShowOpenDialog(
			native.OpenDialogOptions{
				Window:     window,
				Title:      "Open folder",
				Properties: []string{"openDirectory"},
			},
			func(result native.OpenDialogResult, err error) {
				if err == nil && !result.Canceled {
					fmt.Println(result.FilePaths)
				}
			},
		)
	})

}
```
